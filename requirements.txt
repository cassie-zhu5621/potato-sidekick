# ---- core pipeline -------------------------------------------------------
opencv-contrib-python  # includes cv2 modules required by MediaPipe; avoid two cv2 wheels
numpy
google-genai         # Gemini Flash video + text trigger smoke test
python-dotenv        # load local API keys from the ignored .env file
mediapipe            # FaceLandmarker / PoseLandmarker (models in perception/weights/)
torch                # Grounding DINO runtime (MPS on Apple Silicon, CPU fallback)
transformers         # Grounding DINO processor and model
pillow               # image bridge used by the Grounding DINO processor

# ---- the robot -----------------------------------------------------------
pyserial             # the CoreS3 link
feetech-servo-sdk    # SCS0009 bus. NOTE: the pip name is feetech-servo-sdk but
                     # the import name is `scservo_sdk`. `scservo-sdk` does not
                     # exist on PyPI -- installing that gets you nothing.

# ---- speech --------------------------------------------------------------
sounddevice          # push-to-talk capture on the laptop mic
faster-whisper       # transcription. The first call downloads the model, which
                     # takes longer than the state machine's STT timeout -- the
                     # loop warms it at startup so no participant ever waits.

# ---- optional faster detector interface ---------------------------------
ultralytics          # --detector yoloworld / yolo; tested alternative to 1 Hz GDINO
anthropic           # OPTIONAL: only for NOTICEBOT_LLM_PROVIDER=anthropic.
                    # The provider is imported lazily, so leaving this
                    # uninstalled costs nothing on the gemini path.
